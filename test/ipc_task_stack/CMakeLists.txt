foreach(mcu IN ITEMS S3 C3)
  add_executable(IpcTaskStack${mcu}Test IpcTaskStackTest.cpp ${REPO_ROOT}/lib/hal/IpcTaskStack.cpp)
  target_include_directories(IpcTaskStack${mcu}Test PRIVATE stubs)

endforeach()
